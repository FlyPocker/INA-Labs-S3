-- ZADANIE 2 --
CREATE TABLE Producent (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nazwa VARCHAR(50) NOT NULL,
    kraj VARCHAR(20) DEFAULT 'nieznany',
    adresKorespondencyjny VARCHAR(100) DEFAULT 'nieznany'
);

CREATE TABLE Matryca (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    przekatna DECIMAL(4,2) CHECK (przekatna >= 0),
    rozdzielczosc DECIMAL(3,1) CHECK (rozdzielczosc >= 0),
    typ VARCHAR(10)
) AUTO_INCREMENT = 100;

CREATE TABLE Obiektyw (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    model VARCHAR(30),
    minPrzeslona FLOAT CHECK (minPrzeslona >= 0),
    maxPrzeslona FLOAT CHECK (maxPrzeslona >= 0),
    CHECK (minPrzeslona < maxPrzeslona)
);

CREATE TABLE Aparat (
    model VARCHAR(30) PRIMARY KEY,
    producent INT,
    matryca INT,
    obiektyw INT,
    waga FLOAT CHECK (waga >= 0),
    typ ENUM('kompaktowy', 'lustrzanka', 'profesjonalny', 'inny'),
    FOREIGN KEY (producent) REFERENCES Producent(ID),
    FOREIGN KEY (matryca) REFERENCES Matryca(ID),
    FOREIGN KEY (obiektyw) REFERENCES Obiektyw(ID)
);

-- ZADANIE 4 --
/*
1.Uzytkownik nie jest w stanie stworzyc tej procedury ponieważ nie posiada on uprawnienia
    CREATE ROUTINE oraz INSERT na tabele aparat aby procedura rowniez mogla wstawiac do niej dane
2.Uzytkownik jest w stanie wykonać procedure ponieważ ma uprawnienia EXECUTE, aby to zrobic musi wywolac:
    CALL GenerujAparaty();
*/
DELIMITER $$
CREATE PROCEDURE GenerujAparaty()
BEGIN
    DECLARE i INT DEFAULT 0;
    DECLARE max_nr INT DEFAULT 0;
    DECLARE p_id INT;
    DECLARE m_id INT;
    DECLARE o_id INT;

    SELECT COALESCE(MAX(CAST(SUBSTRING(model, 8) AS UNSIGNED)), 0) INTO max_nr 
    FROM Aparat 
    WHERE model LIKE 'Aparat-%';
    
    WHILE i < 100 DO
        SET max_nr = max_nr + 1;
        
        SELECT ID INTO p_id FROM Producent ORDER BY RAND() LIMIT 1;
        SELECT ID INTO m_id FROM Matryca ORDER BY RAND() LIMIT 1;
        SELECT ID INTO o_id FROM Obiektyw ORDER BY RAND() LIMIT 1;

        INSERT INTO Aparat (model, producent, matryca, obiektyw, waga, typ)
        VALUES (CONCAT('Aparat-', max_nr), p_id, m_id, o_id, 0.5, 'kompaktowy');

        SET i = i + 1;
    END WHILE;
END$$
DELIMITER ;

-- ZADANIE 5 --
/*
Konsekwencje ustawienia funkcji na DETERMINISTIC oznaczaja dla bazy danych, ze 
funkcja zwraca dla podanego argumentu zawsze to samo, wtedy baza dostaje opcje 
robnienia cache wyniku funkcji dla konkretnych danych.
Jednak w tym przypadku nie mozemy tej funkcji ustawic na deterministyczna, poniewaz
mogloby to powodowac bledne wyniki w momencie dodania nowych aparatow do tabeli aparaty
funkcja czyta dane z tabeli wiec nie powinna byc deterministyczna.
*/

DELIMITER $$
CREATE FUNCTION ModelNajmniejszaMatryca(id_prod INT)
RETURNS VARCHAR(30)
NOT DETERMINISTIC 
READS SQL DATA
BEGIN
    DECLARE nazwa_modelu VARCHAR(30);

    SELECT a.model INTO nazwa_modelu
    FROM Aparat a
    JOIN Matryca m ON a.matryca = m.ID
    WHERE a.producent = id_prod
    ORDER BY m.przekatna ASC
    LIMIT 1;

    RETURN nazwa_modelu;
END$$
DELIMITER ;

-- ZADANIE 6 --
DELIMITER $$
CREATE FUNCTION sprawdz_aparat(id_producenta INT, id_matrycy INT, id_obiektywu INT)
RETURNS BOOLEAN
READS SQL DATA
BEGIN
    DECLARE prod_istnieje INT;
    DECLARE mat_istnieje INT;
    DECLARE ob_istnieje INT;
    SELECT COUNT(*) INTO prod_istnieje FROM Producent WHERE ID = id_producenta;
    SELECT COUNT(*) INTO mat_istnieje FROM Matryca WHERE ID = id_matrycy;
    SELECT COUNT(*) INTO ob_istnieje FROM Obiektyw WHERE ID = id_obiektywu;

    IF prod_istnieje > 0 AND mat_istnieje > 0 AND ob_istnieje > 0 THEN
        RETURN TRUE;
    ELSE
        RETURN FALSE;
    END IF;
END$$
DELIMITER ;

/*Tutaj musimy stworzyc dwa wyzwalacze poniewaz MySQL, chyba rowniez jak MariaDB
nie pozwalaja na nasluchiwanie dwoch roznych akcji w triggerze*/
DELIMITER $$
CREATE TRIGGER BlokujEdycjeObiektywu BEFORE UPDATE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE czy_przypisany INT;
    SELECT COUNT(*) INTO czy_przypisany FROM Aparat WHERE obiektyw = OLD.ID;
    
    IF czy_przypisany > 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Błąd: Ten obiektyw jest przypisany do aparatu i nie moze byc modyfikowany.';
    END IF;
END$$

CREATE TRIGGER BlokujUsuniecieObiektywu BEFORE DELETE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE czy_przypisany INT;
    SELECT COUNT(*) INTO czy_przypisany FROM Aparat WHERE obiektyw = OLD.ID;
    
    IF czy_przypisany > 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Błąd: Ten obiektyw jest przypisany do aparatu i nie moze byc usuniety.';
    END IF;
END$$
DELIMITER ;

-- ZADANIE 8 --
CREATE TABLE HistoriaProducentow (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_producenta INT,
    stary_kraj VARCHAR(20),
    nowy_kraj VARCHAR(20),
    data_zmiany DATETIME DEFAULT CURRENT_TIMESTAMP
);

DELIMITER $$
CREATE TRIGGER ZmianaKrajuProducenta AFTER UPDATE ON Producent
FOR EACH ROW
BEGIN
    IF OLD.kraj != NEW.kraj THEN
        INSERT INTO HistoriaProducentow (id_producenta, stary_kraj, nowy_kraj, data_zmiany)
        VALUES (OLD.ID, OLD.kraj, NEW.kraj, NOW());
    END IF;
END$$
DELIMITER ;
