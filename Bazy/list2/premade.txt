/* =========================================================================
ZADANIE 2 (Zadanie 2 z listy 2)
Utwórz tabele o schematach:
1. Aparat (model: varchar(30), producent: int, matryca: int, obiektyw: int, 
   waga: float, typ: enum('kompaktowy', 'lustrzanka', 'profesjonalny', 'inny'))
2. Matryca (ID: int, przekatna: decimal(4,2), rozdzielczosc: decimal(3,1), typ: varchar(10))
3. Obiektyw (ID: int, model: varchar(30), minPrzeslona: float, maxPrzeslona: float)
4. Producent (ID: int, nazwa: varchar(50), kraj: varchar(20), adresKorespondencyjny: varchar(100))
Zadbaj o to, by podkreślone atrybuty były kluczami głównymi, tam gdzie to możliwe zastosuj automatyczną 
inkrementację, w przypadku tabeli Matryca zacznij od ID o wartości 100. W tabeli Aparat zadbaj by odpowiednie 
identyfikatory były kluczami obcymi. Dodatkowo zadbaj, aby wprowadzane dane liczbowe nie miały wartości ujemnych 
a minPrzeslona < maxPrzeslona. W tabeli Producent nazwa producenta nie może być pusta, a kraj oraz adres 
korespondencyjny dla siedziby głównej producenta powinny być ustawione na nieznany w przypadku braku danych.
========================================================================= */

CREATE TABLE Producent (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nazwa VARCHAR(50) NOT NULL,
    kraj VARCHAR(20) DEFAULT 'nieznany',
    adresKorespondencyjny VARCHAR(100) DEFAULT 'nieznany'
);

CREATE TABLE Matryca (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    przekatna DECIMAL(4,2) CHECK (przekatna >= 0),
    rozdzielczosc DECIMAL(3,1) CHECK (rozdzielczosc >= 0),
    typ VARCHAR(10)
) AUTO_INCREMENT = 100;

CREATE TABLE Obiektyw (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    model VARCHAR(30),
    minPrzeslona FLOAT CHECK (minPrzeslona >= 0),
    maxPrzeslona FLOAT CHECK (maxPrzeslona >= 0),
    CHECK (minPrzeslona < maxPrzeslona)
);

CREATE TABLE Aparat (
    model VARCHAR(30) PRIMARY KEY,
    producent INT,
    matryca INT,
    obiektyw INT,
    waga FLOAT CHECK (waga >= 0),
    typ ENUM('kompaktowy', 'lustrzanka', 'profesjonalny', 'inny'),
    FOREIGN KEY (producent) REFERENCES Producent(ID),
    FOREIGN KEY (matryca) REFERENCES Matryca(ID),
    FOREIGN KEY (obiektyw) REFERENCES Obiektyw(ID)
);

SHOW FULL TABLES;

ZADANIE 4
Zmodyfikuj zadanie 4 z listy tak żeby nazwy generowanych aparatów miały kształt 
Aparat-nr gdzie numer nadawany jest sekwencyjnie. Zapewnij możliwość wielekrotnego 
wywoływania procedury bez błędów zgłaszanych przez silnik.
Czy użytkownik nralbumu może utworzyć tę procedurę? Napisz jakie uprawnienie mu 
na to pozwala. albo jeżeli nie może, to jakie uprawnienie musiałby mieć, żeby utworzyć 
taką procedurę? Czy może ją wykonać? Podaj komendę.
========================================================================= */

DELIMITER $$
CREATE PROCEDURE GenerujAparatySekwencyjnie()
BEGIN
    DECLARE i INT DEFAULT 0;
    DECLARE max_nr INT DEFAULT 0;
    DECLARE p_id INT;
    DECLARE m_id INT;
    DECLARE o_id INT;

    -- Znajdujemy aktualnie najwyższy numer aparatu w bazie, by kontynuować sekwencję (Aparat-1, Aparat-2...)
    -- COALESCE chroni przed błędem, gdy tabela jest jeszcze pusta (zwróci 0).
    SELECT COALESCE(MAX(CAST(SUBSTRING(model, 8) AS UNSIGNED)), 0) INTO max_nr 
    FROM Aparat 
    WHERE model LIKE 'Aparat-%';
    
    WHILE i < 100 DO
        SET max_nr = max_nr + 1;
        
        -- Losowanie kluczy obcych z innych tabel
        SELECT ID INTO p_id FROM Producent ORDER BY RAND() LIMIT 1;
        SELECT ID INTO m_id FROM Matryca ORDER BY RAND() LIMIT 1;
        SELECT ID INTO o_id FROM Obiektyw ORDER BY RAND() LIMIT 1;

        INSERT INTO Aparat (model, producent, matryca, obiektyw, waga, typ)
        VALUES (CONCAT('Aparat-', max_nr), p_id, m_id, o_id, 0.5, 'kompaktowy');

        SET i = i + 1;
    END WHILE;
END$$
DELIMITER ;

START TRANSACTION;

-- Wywołanie procedury (załóżmy, że nazwałeś ją generuj_aparaty i przyjmuje liczbę sztuk)
CALL GenerujAparatySekwencyjnie(); 

-- Sprawdzenie, czy aparaty (Aparat-1, Aparat-2 itd.) się dodały
SELECT * FROM Aparat;

-- Cofnięcie zmian - baza wraca do stanu sprzed wykonania CALL
ROLLBACK;

/* =========================================================================
ZADANIE 5
Zmień procedurę z zadania 5 z listy na funkcję. 
Napisz jakie są konsekwencję ustawienia tej funkcji jako deterministycznej?
========================================================================= */

DELIMITER $$
CREATE FUNCTION ModelNajmniejszaMatryca_F(id_prod INT)
RETURNS VARCHAR(30)
NOT DETERMINISTIC 
READS SQL DATA
BEGIN
    DECLARE nazwa_modelu VARCHAR(30);

    SELECT a.model INTO nazwa_modelu
    FROM Aparat a
    JOIN Matryca m ON a.matryca = m.ID
    WHERE a.producent = id_prod
    ORDER BY m.przekatna ASC
    LIMIT 1;

    RETURN nazwa_modelu;
END$$
DELIMITER ;

/* ROZWIĄZANIE CZĘŚCI TEORETYCZNEJ (ZADANIE 5):
Konsekwencje ustawienia tej funkcji jako DETERMINISTIC:
Ustawienie funkcji jako DETERMINISTIC byłoby w tym przypadku błędem (okłamaniem silnika bazy). 
Mówi ono bazie, że dla konkretnego "id_prod" funkcja ZAWSZE zwróci ten sam model aparatu, co pozwoli 
jej zbuforować (zapamiętać) wynik. 
Jeśli w przyszłości do tabeli zostanie dodany nowy aparat tego samego producenta z jeszcze mniejszą matrycą, 
funkcja mogłaby zwracać stary (nieaktualny) wynik z cache'u zamiast pobrać go na nowo z tabeli. 
Dodatkowo mogłoby to spowodować błędy w środowiskach z replikacją (rozjazd danych między master a slave).
*/


/* =========================================================================
ZADANIE 6
Napisz funkcję sprawdź_aparat(id_producenta INT, id_matrycy INT, id_obiektywu INT), 
która sprawdza, czy wszystkie trzy powiązane klucze obce tabeli aparat - producent, 
matryca, aparat podane jako argument funkcji, istnieją w odpowiadających im tabelach. 
Funkcja powinna zwracać true albo false.
========================================================================= */

DELIMITER $$
CREATE FUNCTION sprawdz_aparat(id_producenta INT, id_matrycy INT, id_obiektywu INT)
RETURNS BOOLEAN
READS SQL DATA
BEGIN
    DECLARE prod_istnieje INT;
    DECLARE mat_istnieje INT;
    DECLARE ob_istnieje INT;

    SELECT COUNT(*) INTO prod_istnieje FROM Producent WHERE ID = id_producenta;
    SELECT COUNT(*) INTO mat_istnieje FROM Matryca WHERE ID = id_matrycy;
    SELECT COUNT(*) INTO ob_istnieje FROM Obiektyw WHERE ID = id_obiektywu;

    IF prod_istnieje > 0 AND mat_istnieje > 0 AND ob_istnieje > 0 THEN
        RETURN TRUE;
    ELSE
        RETURN FALSE;
    END IF;
END$$
DELIMITER ;

Napisz wyzwalacz blukujący edycję (usunięcie lub modyfikację) rekordu w tabeli obiektyw, 
jeśli dany obiektyw jest przypisany do chociaż jednego aparatu z tabeli aparat. 
Wyzwalacz powinien zgłosić czytelny błąd użytkownikowi (np. poprzez 
SIGNAL SQLSTATE '45000' z odpowiednim komunikatem MESSAGE_TEXT).
========================================================================= */
-- UWAGA: W MySQL/MariaDB nie można zrobić jednego triggera nasłuchującego na dwie 
-- różne akcje naraz. Należy utworzyć osobny trigger dla UPDATE i osobny dla DELETE.

DELIMITER $$
CREATE TRIGGER BlokujEdycjeObiektywu_UPDATE BEFORE UPDATE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE czy_przypisany INT;
    SELECT COUNT(*) INTO czy_przypisany FROM Aparat WHERE obiektyw = OLD.ID;
    
    IF czy_przypisany > 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Błąd: Ten obiektyw jest przypisany do aparatu i nie moze byc modyfikowany.';
    END IF;
END$$

CREATE TRIGGER BlokujEdycjeObiektywu_DELETE BEFORE DELETE ON Obiektyw
FOR EACH ROW
BEGIN
    DECLARE czy_przypisany INT;
    SELECT COUNT(*) INTO czy_przypisany FROM Aparat WHERE obiektyw = OLD.ID;
    
    IF czy_przypisany > 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Błąd: Ten obiektyw jest przypisany do aparatu i nie moze byc usuniety.';
    END IF;
END$$
DELIMITER ;

START TRANSACTION;

-- Próba usunięcia obiektywu (podmień '1' na ID obiektywu, który JEST przypisany do jakiegoś aparatu)
DELETE FROM Obiektyw  WHERE id = 1; 
-- Oczekiwany wynik: Błąd SQLSTATE 45000 i komunikat zdefiniowany w wyzwalaczu

-- Próba modyfikacji zablokowanego obiektywu
UPDATE Obiektyw SET model = 'Test' WHERE id = 1;
-- Oczekiwany wynik: Błąd SQLSTATE 45000

ROLLBACK;

/* =========================================================================
ZADANIE 8
Stwórz tabelę HistoriaProducentów (id, id_producenta, stary_kraj, nowy_kraj, data_zmiany). 
Napisz wyzwalacz na tabeli Producent, który przy każdej zmianie wartości kraj 
automatycznie zarejestruję zmianę w tabeli histori.
========================================================================= */

CREATE TABLE HistoriaProducentow (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_producenta INT,
    stary_kraj VARCHAR(20),
    nowy_kraj VARCHAR(20),
    data_zmiany DATETIME DEFAULT CURRENT_TIMESTAMP
);

DELIMITER $$
CREATE TRIGGER ZmianaKrajuProducenta AFTER UPDATE ON Producent
FOR EACH ROW
BEGIN
    -- Uruchamiamy logowanie tylko i wyłącznie, gdy faktycznie zmienił się wpis w kolumnie 'kraj'
    IF OLD.kraj != NEW.kraj THEN
        INSERT INTO HistoriaProducentow (id_producenta, stary_kraj, nowy_kraj, data_zmiany)
        VALUES (OLD.ID, OLD.kraj, NEW.kraj, NOW());
    END IF;
END$$
DELIMITER ;

START TRANSACTION;

-- Zmiana kraju dla przykładowego producenta (np. ID 1)
UPDATE Producent SET kraj = 'Nowy Kraj Testowy' WHERE id = 1;

-- Sprawdzenie, czy trigger zarejestrował zmianę w tabeli historii
SELECT * FROM HistoriaProducentow; 
-- Oczekiwany wynik: Wiersz zawierający stare ID, stary kraj, 'Nowy Kraj Testowy' oraz datę.

-- Cofnięcie zmiany (kraj producenta wraca do normy, a wpis z historii znika)
ROLLBACK;
